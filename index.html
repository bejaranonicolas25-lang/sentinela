// sw.js - Hace que SENTINELA sea una APP offline const CACHE = "sentinela-v2"; const ARCHIVOS = [ "./", "./index.html", "./style.css", "./app.js", "./fondo-leon.jpg", "./manifest.json" ]; self.addEventListener('install', e => { e.waitUntil(caches.open(CACHE).then(c => c.addAll(ARCHIVOS))); }); self.addEventListener('fetch', e => { // Si no hay internet, usa lo guardado. Si hay, usa internet. e.respondWith( caches.match(e.request).then(res => res || fetch(e.request)) ); });
