// VOZ SUAVE SIN CORTES - CENTINELA let vozLista = null; function cargarVoz(){ try{ let voces = speechSynthesis.getVoices(); // Busca la mejor voz en español, la más natural vozLista = voces.find(v=>v.lang==='es-AR' && v.name.toLowerCase().includes('google')) || voces.find(v=>v.lang.startsWith('es')) || voces[0]; }catch(e){} } if('speechSynthesis' in window){ cargarVoz(); speechSynthesis.onvoiceschanged = cargarVoz; } function hablar(texto){ try{ speechSynthesis.cancel(); let t = texto || 'Alfa Bravo diez cuatro, Centinela activo, Nico Bejarano'; // Lo partimos en pedacitos cortos para que no se corte let partes = t.match(/[^.!?]+[.!?]+|[^.!?]+$/g) || [t]; let i = 0; function decirSiguiente(){ if(i >= partes.length) return; let u = new SpeechSynthesisUtterance(partes[i].trim()); u.lang = 'es-AR'; u.rate = 0.92; // más lento = no se traba u.pitch = 0.95; u.volume = 1; if(vozLista) u.voice = vozLista; u.onend = function(){ i++; setTimeout(decirSiguiente, 120); }; u.onerror = function(){ i++; setTimeout(decirSiguiente, 120); }; speechSynthesis.speak(u); } // Esperamos 150ms a que se limpie la voz anterior setTimeout(decirSiguiente, 150); }catch(e){} } function pararVoz(){ try{ speechSynthesis.cancel(); }catch(e){} }
